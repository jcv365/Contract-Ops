import { z } from 'zod';
 
// z.coerce.boolean() treats any non-empty string as true — "false" would
// enable ENABLE_GRAPH_SEND. This only accepts the literal strings.
const strictBoolean = (defaultValue: boolean) =>
  z
    .enum(['true', 'false'])
    .default(defaultValue ? 'true' : 'false')
    .transform((v) => v === 'true');
 
// Environment configuration schema
export const envSchema = z.object({
  // Server
  NODE_ENV: z.enum(['development', 'production', 'test']).default('development'),
  PORT: z.coerce.number().default(3000),
  HOST: z.string().default('0.0.0.0'),
 
  // Database
  DATABASE_URL: z.string().url(),
 
  // Redis
  REDIS_URL: z.string().url().default('redis://localhost:6379'),
 
  // Microsoft Graph / Entra ID
  AZURE_TENANT_ID: z.string().optional(),
  AZURE_CLIENT_ID: z.string().optional(),
  AZURE_CLIENT_SECRET: z.string().optional(),
  // docker-compose's ${VAR:-} interpolation yields an empty string, not an
  // absent key, when the .env value is blank — z.string().url().optional()
  // rejects "" (it's a defined value that fails .url(), not `undefined`),
  // which crashed the container on first deploy. Treat blank as absent.
  GRAPH_REDIRECT_URI: z
    .union([z.literal(''), z.string().url()])
    .optional()
    .transform((v) => (v ? v : undefined)),
 
  // AI Provider — routed through freellmapi (CLAUDE.md rule 4: reuse
  // existing model routing/infrastructure rather than adding a fourth AI
  // vendor). openai/anthropic/local are kept only as an escape hatch for
  // local dev against a different router; freellmapi is the default and
  // the only path this build actually exercises.
  AI_PROVIDER: z.enum(['freellmapi', 'openai', 'anthropic', 'local']).default('freellmapi'),
  FREELLMAPI_BASE_URL: z.string().url().default('http://host.docker.internal:3010'),
  FREELLMAPI_API_KEY: z.string().optional(),
  OPENAI_API_KEY: z.string().optional(),
  ANTHROPIC_API_KEY: z.string().optional(),
  AI_MODEL: z.string().default('gpt-4o-mini'),
 
  // Search/Scraping
  USER_AGENT: z.string().default('HUNT-Contract-Acquisition-OS/1.0'),
  REQUEST_TIMEOUT_MS: z.coerce.number().default(30000),
  MAX_CONCURRENT_REQUESTS: z.coerce.number().default(5),
 
  // Rate Limiting
  RATE_LIMIT_WINDOW_MS: z.coerce.number().default(900000), // 15 minutes
  RATE_LIMIT_MAX_REQUESTS: z.coerce.number().default(100),
 
  // Logging
  LOG_LEVEL: z.enum(['trace', 'debug', 'info', 'warn', 'error', 'fatal']).default('info'),
  LOG_PRETTY: strictBoolean(true),
 
  // Security
  JWT_SECRET: z.string().min(32).optional(),
  ENCRYPTION_KEY: z.string().length(32).optional(),
  // docs-pack/docs/14/28: least privilege. This is a single-user personal
  // system (rule 18: Phase 1 explicitly does not build multi-tenant/SaaS
  // auth) — a shared-secret API key is the proportionate access control
  // for the JSON API: without it, nothing on the API surface (other than
  // /health) is reachable at all. Optional only so local dev without a
  // key still works; startup preflight warns loudly when it's unset in a
  // way that implies production use.
  API_KEY: z.string().min(16).optional(),
 
  // Admin portal (src/portal/): a single password login, session-based,
  // separate from API_KEY — the portal is a human at a browser, the API
  // key is for programmatic callers. Same "required in production"
  // treatment.
  PORTAL_PASSWORD: z.string().min(8).optional(),
  SESSION_SECRET: z.string().min(32).optional(),
 
  // Feature Flags — safety-critical: must fail closed, never coerce a
  // truthy-string bug into enabling autonomous sending (ADR-002).
  ENABLE_REPLY_INGESTION: strictBoolean(false),
  ENABLE_GRAPH_SEND: strictBoolean(false),
  ENABLE_BIDOPS: strictBoolean(true),
 
  // Web-fetch discovery (docs/22 — org-website contact discovery, Sprint
  // E). Defaults false: fetching arbitrary third-party sites is a
  // materially different risk from calling a documented job-board API, so
  // it stays opt-in even though the compliance layer (web-fetch/) fails
  // closed regardless.
  ENABLE_WEB_DISCOVERY: strictBoolean(false),
  WEB_FETCH_MAX_PAGES_PER_COMPANY: z.coerce.number().default(12),
  WEB_FETCH_DEFAULT_RPM: z.coerce.number().default(6),
  // rule 14: every outbound request must identify itself honestly — for
  // web-fetch specifically (as opposed to the job-board adapters' generic
  // USER_AGENT) that means a real URL a site operator can visit to see who
  // is requesting their pages and why. Required before discovery can run
  // at all — see preflight.ts.
  WEB_FETCH_CONTACT_URL: z.string().url().optional(),
 
  // Attachment storage (modules/attachments/service.ts) — a local
  // filesystem path, backed by a named Docker volume in docker-compose.yml
  // so uploads survive a container recreate.
  ATTACHMENT_STORAGE_DIR: z.string().default('/app/storage/attachments'),
 
  // Search-based organisation discovery (modules/company-discovery/) —
  // finds vendor partners / regional recruiters via a web search API,
  // distinct from ENABLE_WEB_DISCOVERY (which only crawls a company's OWN
  // site once it already exists as a Company row). Same fail-closed
  // default as every other discovery flag.
  ENABLE_SEARCH_DISCOVERY: strictBoolean(false),
  BRAVE_SEARCH_API_KEY: z.string().optional(),
});
 
export type Env = z.infer<typeof envSchema>;
 
let cachedEnv: Env | null = null;
 
export function getEnv(): Env {
  if (cachedEnv) return cachedEnv;
 
  const result = envSchema.safeParse(process.env);
  if (!result.success) {
    console.error('Invalid environment configuration:', result.error.flatten().fieldErrors);
    throw new Error('Invalid environment configuration');
  }
 
  cachedEnv = result.data;
  return cachedEnv;
}
 
export const env = getEnv();
